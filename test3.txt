"""
Updated plotting functions for consistent map styling.

This module provides updated versions of the `plot_data_map` and
`plot_all_data_maps` functions with improved styling consistent across
all plots. It also includes helper functions for preparing grids,
adding land/ocean backgrounds, adding region labels, and drawing
circular boundaries for polar projections.

These functions are designed to work with xarray.DataArray objects
representing spatial fields. They handle both regular (1D lat/lon) and
curvilinear (2D latitude/longitude) grids, apply appropriate map
projections, and produce maps with a unified look suitable for
publication-quality figures.
"""

import os
from typing import Dict, Any

import numpy as np
import xarray as xr
import matplotlib.pyplot as plt
import matplotlib.colors as mcolors
import cartopy.crs as ccrs
import cartopy.feature as cfeature
from matplotlib.path import Path

__all__ = [
    "_prepare_lon_lat",
    "_circular_polar_boundary",
    "_add_region_labels",
    "_add_land_ocean",
    "plot_data_map",
    "plot_all_data_maps",
]


def _prepare_lon_lat(da: xr.DataArray) -> tuple[np.ndarray, np.ndarray]:
    """Return 2D arrays of longitude and latitude.

    Handles both regular grids with 1D ``lat``/``lon`` coordinates and
    curvilinear grids with 2D ``latitude``/``longitude`` coordinates.

    Parameters
    ----------
    da : xarray.DataArray
        Input data array.

    Returns
    -------
    lon2d, lat2d : numpy.ndarray
        Two-dimensional arrays of longitude and latitude.

    Raises
    ------
    ValueError
        If the data array does not contain recognizable coordinate names.
    """
    dims = set(da.dims)
    if {"lat", "lon"}.issubset(dims):
        lon2d, lat2d = np.meshgrid(da.lon, da.lat)
    elif {"j", "i"}.issubset(dims) and {"latitude", "longitude"}.issubset(da.coords):
        lat2d, lon2d = da.latitude, da.longitude
    else:
        raise ValueError(
            "Cannot determine latitude/longitude coordinates. Expected 'lat'/'lon' or 'latitude'/'longitude' with 'i'/'j'."
        )
    return lon2d, lat2d


def _circular_polar_boundary(ax: plt.Axes, radius: float = 0.50) -> None:
    """Apply a circular boundary to polar maps.

    For polar stereographic maps, this helper draws a circular boundary
    centered in the axes coordinate system. Without centering the circle
    properly, the map may be clipped incorrectly.

    Parameters
    ----------
    ax : matplotlib.axes.Axes
        The axes to which the boundary will be applied.
    radius : float, optional
        Radius of the circle in axes coordinates (default is 0.50).

    Notes
    -----
    This function modifies the axes in-place and does not return
    anything.
    """
    theta = np.linspace(0, 2 * np.pi, 361)
    center_x, center_y = 0.5, 0.5
    verts = np.vstack([
        center_x + radius * np.sin(theta),
        center_y + radius * np.cos(theta),
    ]).T
    ax.set_boundary(Path(verts), transform=ax.transAxes)


def _add_region_labels(
    ax: plt.Axes,
    region: str,
    transform: ccrs.Projection,
    base_fontsize: float = 5.0,
    marker_size: float = 0.5,
    marker_color: str = "black",
) -> None:
    """Add area names and markers for selected regions.

    This helper annotates maps with short region names and small
    markers. Font sizes scale with name length to avoid clutter, and
    markers indicate the exact location of each label.

    Parameters
    ----------
    ax : matplotlib.axes.Axes
        Axes on which to draw the labels.
    region : str
        One of 'arctic', 'antarctic', or 'global'. Determines which
        labels to display.
    transform : cartopy.crs.Projection
        Coordinate transform for placing markers and text.
    base_fontsize : float, optional
        Base font size for short labels. Longer labels are scaled
        down (default 5.0).
    marker_size : float, optional
        Size of the circular marker indicating each label location
        (default 0.5).
    marker_color : str, optional
        Colour of the markers (default 'black').
    """
    region = region.lower()
    # Define label locations (lon, lat). Keys are shortened names.
    if region == "arctic":
        labels = {
            "Bea": (-150, 73),
            "Chu": (-168, 69),
            "Eas": (155, 73),
            "Lap": (125, 76),
            "Kar": (75, 73),
            "Bar": (40, 74),
            "Gre": (-15, 75),
            "Baf": (-65, 70),
            "Hud": (-90, 59),
            "Nor": (0, 71),
            "Arctic": (0, 88),
        }
    elif region == "antarctic":
        labels = {
            "Weddell Sea": (-45, -73),
            "Ross Sea": (175, -75),
            "Amundsen Sea": (-115, -72),
            "Bellingshausen Sea": (-75, -70),
            "Scotia Sea": (-40, -56),
            "Southern Ocean": (0, -62),
        }
    else:
        labels = {
            "Pacific Ocean": (-140, 0),
            "Atlantic Ocean": (-30, -10),
            "Indian Ocean": (85, -15),
            "Arctic Ocean": (0, 80),
            "Southern Ocean": (20, -55),
        }
    for name, (lon, lat) in labels.items():
        ax.plot(
            lon,
            lat,
            marker="o",
            markersize=marker_size,
            markeredgecolor=marker_color,
            markerfacecolor=marker_color,
            transform=transform,
            zorder=5,
        )
        length = len(name)
        if length <= 12:
            fontsize = base_fontsize
        elif length <= 20:
            fontsize = base_fontsize - 1
        else:
            fontsize = base_fontsize - 2
        ax.text(
            lon + 3,
            lat + 3,
            name,
            transform=transform,
            fontsize=fontsize,
            ha="left",
            va="bottom",
            zorder=10,
        )


def _add_land_ocean(
    ax: plt.Axes, land_color: str = "white", ocean_color: str = "#D6ECFF"
) -> None:
    """Draw land and ocean backgrounds with consistent colours.

    Paints the ocean first and then overlays land, borders, and
    coastlines. This ensures land appears clean and white, and ocean
    appears soft blue, regardless of the data plotted.

    Parameters
    ----------
    ax : matplotlib.axes.Axes
        Axes on which to draw the map features.
    land_color : str, optional
        Colour for land areas (default 'white').
    ocean_color : str, optional
        Colour for ocean areas (default '#D6ECFF').
    """
    ax.set_facecolor(ocean_color)
    ax.add_feature(
        cfeature.OCEAN.with_scale("50m"),
        facecolor=ocean_color,
        zorder=0,
    )
    land = cfeature.LAND.with_scale("50m")
    ax.add_feature(
        land,
        facecolor=land_color,
        edgecolor="0.5",
        linewidth=0.3,
        zorder=3,
    )
    ax.coastlines(resolution="50m", linewidth=0.4, color="0.3", zorder=4)
    ax.add_feature(
        cfeature.BORDERS.with_scale("50m"),
        linewidth=0.3,
        edgecolor="0.35",
        zorder=4,
    )


def plot_data_map(
    data: xr.DataArray,
    model: str,
    region: str,
    variable_name: str = "Variable",
    save_dir: str = "pic",
    cmap: str = "turbo",
    unit: str = "",
    experiment: str = "",
    smooth: bool = False,
    dpi: int = 300,
    land_color: str = "white",
    ocean_color: str = "#D6ECFF",
    add_area_names: bool = True,
    annotate_source: bool = False,
) -> None:
    """Draw a single map with consistent styling.

    Parameters
    ----------
    data : xarray.DataArray
        The data to plot. Must be at least 2D and contain lat/lon
        coordinates. If it has a time dimension, it will be averaged
        before plotting.
    model : str
        Name of the model (for titles).
    region : str
        One of 'Arctic', 'Antarctic', or any other string treated as
        global.
    variable_name : str, optional
        Name of the variable to display (default 'Variable').
    save_dir : str, optional
        Directory to save the plot (default 'pic').
    cmap : str, optional
        Colormap name (default 'turbo').
    unit : str, optional
        Unit for the colourbar label (default '').
    experiment : str, optional
        Experiment name (appended in title and filename).
    smooth : bool, optional
        Whether to use Gouraud shading (default False).
    dpi : int, optional
        Resolution of the saved figure (default 300).
    land_color : str, optional
        Colour for land areas (default 'white').
    ocean_color : str, optional
        Colour for ocean areas (default '#D6ECFF').
    add_area_names : bool, optional
        Whether to draw region labels (default True).
    annotate_source : bool, optional
        Whether to add a small annotation with projection information
        (default False).

    Notes
    -----
    This function automatically handles regular and curvilinear grids,
    chooses an appropriate projection based on the region, and masks
    invalid data. It saves the figure to the specified directory.
    """
    if isinstance(data, dict):
        raise ValueError("`data` must be an xarray.DataArray, not a dict")
    # Average over time if necessary
    if "time" in data.dims:
        data = data.mean("time")
    lon2d, lat2d = _prepare_lon_lat(data)
    # Compute masked array
    masked_data = np.ma.masked_invalid(data.compute())
    if masked_data.mask.all():
        print(f"⚠ {model}/{experiment}/{region}: only NaNs → skip plot")
        return
    # Choose projection
    reg = region.lower()
    projection = {
        "arctic": ccrs.NorthPolarStereo(),
        "antarctic": ccrs.SouthPolarStereo(),
    }.get(reg, ccrs.PlateCarree())
    fig = plt.figure(figsize=(9, 5))
    ax = plt.axes(projection=projection)
    # Set extent and boundary for polar maps
    if reg == "global":
        ax.set_global()
    elif reg == "arctic":
        ax.set_extent([-180, 180, 50, 90], crs=ccrs.PlateCarree())
        _circular_polar_boundary(ax)
    elif reg == "antarctic":
        ax.set_extent([-180, 180, -90, -50], crs=ccrs.PlateCarree())
        _circular_polar_boundary(ax)
    # Draw land and ocean backgrounds
    _add_land_ocean(ax, land_color=land_color, ocean_color=ocean_color)
    # Plot data
    shade_kw = dict(shading="gouraud") if smooth else dict(shading="auto")
    data_zorder = 2
    if np.nanmin(masked_data) >= 0:
        mesh = ax.pcolormesh(
            lon2d,
            lat2d,
            masked_data,
            cmap=cmap,
            transform=ccrs.PlateCarree(),
            zorder=data_zorder,
            **shade_kw,
        )
        cbar = plt.colorbar(
            mesh,
            ax=ax,
            orientation="horizontal",
            pad=0.07,
            shrink=0.55,
        )
    else:
        vmax = np.nanpercentile(np.abs(masked_data), 99)
        norm = mcolors.TwoSlopeNorm(vmin=-vmax, vcenter=0, vmax=vmax)
        mesh = ax.pcolormesh(
            lon2d,
            lat2d,
            masked_data,
            cmap=cmap,
            norm=norm,
            transform=ccrs.PlateCarree(),
            zorder=data_zorder,
            **shade_kw,
        )
        cbar = plt.colorbar(
            mesh,
            ax=ax,
            orientation="horizontal",
            pad=0.07,
            shrink=0.55
        )
    cbar.set_label(f"{variable_name} ({unit})")
    # Gridlines and labels
    if reg == "global":
        gl = ax.gridlines(
            draw_labels=True,
            linewidth=0.5,
            color="gray",
            alpha=0.4,
            linestyle="--",
        )
        gl.right_labels = False
        gl.top_labels = False
    else:
        ax.gridlines(
            draw_labels=False,
            linewidth=0.4,
            color="gray",
            alpha=0.35,
            linestyle="--",
        )
    # Area names
    if add_area_names:
        _add_region_labels(ax, reg, transform=ccrs.PlateCarree())
    # Title
    title = f"{model} – {region} – {variable_name}"
    if experiment:
        title += f" ({experiment})"
    plt.title(title)
    # Optional annotation
    if annotate_source:
        plt.annotate(
            "Projection: Cartopy; Natural Earth 1:50m",
            xy=(0.01, 0.01),
            xycoords="figure fraction",
            fontsize=6,
            alpha=0.7,
        )
    plt.tight_layout()
    # Save figure
    os.makedirs(save_dir, exist_ok=True)
    safe_exp = str(experiment)[:30]
    fname = f"{model}_{region}_{variable_name}_{safe_exp}.png"
    plt.savefig(os.path.join(save_dir, fname.replace(" ", "_")), dpi=dpi)
    plt.close(fig)


def plot_all_data_maps(
    data_dict: Any,
    variable_name: str = "ERF",
    save_dir: str = "images",
    **plot_kwargs: Any,
) -> None:
    """Recursively plot maps for a nested results dictionary.

    Supports two structures:

      - {model: {experiment: {region: DataArray}}}
      - {model: {region: DataArray}}

    If ``data_dict`` is a single DataArray, plots that field.

    Parameters
    ----------
    data_dict : dict or xarray.DataArray
        Input dictionary or single DataArray.
    variable_name : str, optional
        Name of the variable for colourbar label and titles (default 'ERF').
    save_dir : str, optional
        Directory to save plots (default 'images').
    **plot_kwargs : dict
        Additional keyword arguments passed to ``plot_data_map``.

    Notes
    -----
    Invalid or 1D data arrays are skipped. A message is printed
    whenever a field is skipped.
    """
    if isinstance(data_dict, xr.DataArray):
        if np.isnan(data_dict).all() or data_dict.ndim < 2:
            print(
                "⚠ Skipping single DataArray (empty or less than 2 dimensions)"
            )
            return
        plot_data_map(
            data_dict,
            model="unknown",
            region="global",
            variable_name=variable_name,
            save_dir=save_dir,
            **plot_kwargs,
        )
        return
    # Ensure dictionary-like structure
    if not isinstance(data_dict, dict):
        raise ValueError("data_dict must be a dict or a DataArray")
    for model, sub in data_dict.items():
        if not sub:
            continue
        nested = any(isinstance(v, dict) for v in sub.values())
        if nested:
            for exp, reg_dict in sub.items():
                for region, da in reg_dict.items():
                    if not isinstance(da, xr.DataArray):
                        continue
                    if da.ndim < 2 or np.isnan(da).all():
                        continue
                    print(f"Plotting {model} / {exp} / {region}")
                    plot_data_map(
                        da,
                        model=model,
                        region=region,
                        variable_name=variable_name,
                        save_dir=save_dir,
                        experiment=exp,
                        **plot_kwargs,
                    )
        else:
            for region, da in sub.items():
                if not isinstance(da, xr.DataArray):
                    continue
                if da.ndim < 2 or np.isnan(da).all():
                    continue
                print(f"Plotting {model} / {region}")
                plot_data_map(
                    da,
                    model=model,
                    region=region,
                    variable_name=variable_name,
                    save_dir=save_dir,
                    **plot_kwargs,
                )
